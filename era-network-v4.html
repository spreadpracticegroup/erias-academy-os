function showPage(id,btn){
document.querySelectorAll('.page').forEach(p=>p.classList.remove('active'));

if(btn) btn.classList.add('active');
window.scrollTo({top:0,behavior:'smooth'});
}

function envoyerWhatsApp(){
let nom=document.getElementById('nom').value;
let tel=document.getElementById('tel').value;
let prog=document.getElementById('programme').value;
let niv=document.getElementById('niveau').value;
let obj=document.getElementById('objectif').value;
if(!nom||!tel){alert('Stp mets ton nom et ton WhatsApp 🙏');return;}
let msg="Bonjour ERIAS Academy! Je veux rejoindre ERA NETWORK%0A%0A👤 Nom: "+encodeURIComponent(nom)+"%0A📱 WhatsApp: "+encodeURIComponent(tel)+"%0A🎯 Programme: "+encodeURIComponent(prog)+"%0A📊 Niveau: "+encodeURIComponent(niv)+"%0A💬 Objectif: "+encodeURIComponent(obj);
window.open("https://wa.me/24177000000?text="+msg,"_blank");
document.getElementById('okMsg').style.display='block';
}

function corrigerQuiz(){
let score=0; let total=5;
for(let i=1;i<=5;i++){
let r=document.querySelector('input[name="q'+i+'"]:checked');
if(r) score+=parseInt(r.value);
}
let res=document.getElementById('quizResult');
res.style.display='block';
if(score<=2) res.innerHTML="🎯 Ton niveau : DÉBUTANT ("+score+"/5)<br>Parfait pour commencer EERLSZONE ! Clique sur S'inscrire.";
else if(score<=4) res.innerHTML="🔥 Ton niveau : INTERMÉDIAIRE ("+score+"/5)<br>Tu as de bonnes bases ! EERLSZONE + ELDS pour toi.";
else res.innerHTML="🏆 Ton niveau : AVANCÉ ("+score+"/5)<br>Excellent ! ELDS Leadership est fait pour toi.";
}
